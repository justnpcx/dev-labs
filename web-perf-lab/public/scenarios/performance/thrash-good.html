核心改法：把读和写分开

// ✅ 第一步：把所有「读」集中做完
const heights = [];
for (let i = 0; i < boxes.length; i++) {
  heights.push(boxes[i].offsetHeight);   // 只在第一次触发布局
}
// 此时布局已经是最新的，后面所有读都不再触发

// ✅ 第二步：再把所有「写」集中做完
for (let i = 0; i < boxes.length; i++) {
  boxes[i].style.width = (heights[i] * 2) + 'px';
}
// 写操作只让布局「失效」，不会立刻重算 —— 浏览器会攒到帧末统一算一次

如果非要在循环里读怎么办

// 用 rAF 把读写拆到不同帧
requestAnimationFrame(() => {
  const h = el.offsetHeight;            // 这一帧读
  requestAnimationFrame(() => {
    el.style.width = h * 2 + 'px';      // 下一帧写
  });
});

API作用
